public String getShortenedName()
{
    String result = "";
    int i = 0;
    while (i < username.length())
    {
        if (i < username.length() - 1 && username.charAt(i + 1) == '-')
        {
            i += 2;
        }
        else
        {
            result += username.charAt(i);
            i++;
        }
    }
    return result;
}
