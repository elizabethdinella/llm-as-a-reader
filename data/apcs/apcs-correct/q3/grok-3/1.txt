public int moreHistoryThanMathAbsences()
{
    int count = 0;
    for (CourseRecord h : historyList)
    {
        String id = h.getStudentID();
        int hAbs = h.getAbsences();
        for (CourseRecord m : mathList)
        {
            if (m.getStudentID().equals(id))
            {
                if (hAbs > m.getAbsences())
                    count++;
                break;
            }
        }
    }
    return count;
}
