public int getPointsForRow(int targetRow)
{
    int sum = 0;
    boolean same = true;
    String color = board[targetRow][0].getColor();
    for (int c = 0; c < board[0].length; c++)
    {
        Space s = board[targetRow][c];
        sum += s.getPoints();
        if (!s.getColor().equals(color))
            same = false;
    }
    if (same)
        return 2 * sum;
    return sum;
}
