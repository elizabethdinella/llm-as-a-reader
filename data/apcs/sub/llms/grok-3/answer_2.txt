public String getShortenedName()
{
    String result = "";
    for (int i = 0; i < username.length(); i++)
    {
        if (username.charAt(i) == '-')
        {
            result = result.substring(0, result.length() - 1);
        }
        else
        {
            result += username.charAt(i);
        }
    }
    return result;
}
