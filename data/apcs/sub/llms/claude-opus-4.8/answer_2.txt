public String getShortenedName()
{
    String result = "";
    for (int i = 0; i < username.length(); i++)
    {
        char current = username.charAt(i);
        if (current == '-')
        {
            // skip the hyphen itself
            continue;
        }
        if (i + 1 < username.length() && username.charAt(i + 1) == '-')
        {
            // skip the character before a hyphen
            continue;
        }
        result += current;
    }
    return result;
}
