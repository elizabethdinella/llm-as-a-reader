public Account(String requestedName)
{
    if (isAvailable(requestedName))
    {
        username = requestedName;
    }
    else
    {
        int counter = 1;
        String newName = requestedName + counter;
        while (!isAvailable(newName))
        {
            counter++;
            newName = requestedName + counter;
        }
        username = newName;
    }
}
