public int getPointsForRow(int targetRow) {
    int sumPoints;
    String color = board[targetRow][0].getColor();
    boolean sameColor = true;

    for (int col = 0; col < board[targetRow].length; col++) {
        sumPoints += board[targetRow][col].getPoints();
        if (!board[targetRow][col].getColor().equals(color)) {
            sameColor = false;
        }
    }

    if (sameColor) {
        return 2 * sumPoints;
    } else {
        return sumPoints;
    }
}
