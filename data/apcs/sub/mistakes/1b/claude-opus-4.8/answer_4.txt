public String getShortenedName()
{
    String result = "";
    for (int i = 0; i < username.length(); i++)
    {
        if (username.substring(i + 1, i + 2).equals("-"))
        {
            i++;
        }
        else
        {
            result += username.substring(i, i + 1);
        }
    }
    return result;
}
