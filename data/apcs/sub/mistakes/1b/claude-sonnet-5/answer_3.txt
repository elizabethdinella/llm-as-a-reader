public String getShortenedName()
{
    StringBuilder result = new StringBuilder();
    int i = 0;
    while (i < username.length())
    {
        if (username.charAt(i + 1) == '-')
        {
            i += 1;
        }
        else
        {
            result.append(username.charAt(i));
            i++;
        }
    }
    return result.toString();
}
