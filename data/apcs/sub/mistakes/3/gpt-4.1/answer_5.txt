public int moreHistoryThanMathAbsences() {
    int count = 0;
    HashSet<String> mathStudents = new HashSet<>();
    for (CourseRecord cr : mathList) {
        mathStudents.add(cr.getStudentID());
    }
    for (CourseRecord cr : historyList) {
        if (mathStudents.contains(cr.getStudentID())) {
            count++;
        }
    }
    return count;
}
