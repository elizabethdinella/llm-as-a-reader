public void Move(char direction, int minutes) {
    double distance = (DistancePerHour / 60.0) * minutes;
    if (direction == 'N') {
        YPosition += distance;
    } else if (direction == 'S') {
        YPosition -= distance;
    } else if (direction == 'E') {
        XPosition += distance;
    } else if (direction == 'W') {
        XPosition -= distance;
    }
}

```java
import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        Bird bird1 = new Bird("Cockatiel", 71.0);
        Bird bird2 = new Bird("Macaw", 56.0);

        System.out.println(bird1.GetSpecies() + ": " + bird1.GetPosition());
        System.out.println(bird2.GetSpecies() + ": " + bird2.GetPosition());

        Bird selectedBird = null;
        while (selectedBird == null) {
            System.out.print("Select bird (1 for Cockatiel, 2 for Macaw): ");
            int choice = scanner.nextInt();
            if (choice == 1) selectedBird = bird1;
            else if (choice == 2) selectedBird = bird2;
        }

        char direction = ' ';
        while (direction != 'N' && direction != 'S' && direction != 'E' && direction != 'W') {
            System.out.print("Enter direction (N/S/E/W): ");
            direction = scanner.next().toUpperCase().charAt(0);
        }

        int minutes = -1;
        while (minutes < 0 || minutes > 500) {
            System.out.print("Enter minutes flying (0-500): ");
            minutes = scanner.nextInt();
        }

        selectedBird.Move(direction, minutes);
        System.out.println("Updated position: " + selectedBird.GetPosition());
    }
}
```
