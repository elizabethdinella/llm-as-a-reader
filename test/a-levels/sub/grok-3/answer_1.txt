```java
public class Bird {
    private double DistancePerHour;
    private String Species;
    private double XPosition;
    private double YPosition;

    public Bird(String species, double distancePerHour) {
        this.Species = species;
        this.DistancePerHour = distancePerHour;
        this.XPosition = 500.0;
        this.YPosition = 500.0;
    }
}
```

```java
public void Move(char direction, int minutes) {
    double distance = (DistancePerHour / 60.0) * minutes;
    if (direction == 'N') YPosition += distance;
    else if (direction == 'S') YPosition -= distance;
    else if (direction == 'E') XPosition += distance;
    else if (direction == 'W') XPosition -= distance;
}
```

```java
import java.util.Scanner;
Scanner sc = new Scanner(System.in);
Bird b1 = new Bird("Cockatiel", 71.0);
Bird b2 = new Bird("Macaw", 56.0);
System.out.println(b1.GetSpecies() + " " + b1.GetPosition());
System.out.println(b2.GetSpecies() + " " + b2.GetPosition());
Bird sel = null;
while (sel == null) {
    System.out.print("Bird (1/2): ");
    int c = sc.nextInt();
    if (c == 1) sel = b1; else if (c == 2) sel = b2;
}
char dir = 0;
while (dir != 'N' && dir != 'S' && dir != 'E' && dir != 'W') {
    System.out.print("Dir: ");
    dir = sc.next().toUpperCase().charAt(0);
}
int min = -1;
while (min < 0 || min > 500) {
    System.out.print("Mins: ");
    min = sc.nextInt();
}
sel.Move(dir, min);
System.out.println("New: " + sel.GetPosition());
```
