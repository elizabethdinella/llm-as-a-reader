import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner input = new Scanner(System.in);

        Bird bird1 = new Bird("Cockatiel", 71.0);
        Bird bird2 = new Bird("Macaw", 56.0);

        // Output details for each bird
        System.out.println("Bird 1: Species = " + bird1.GetSpecies() + " " + bird1.GetPosition());
        System.out.println("Bird 2: Species = " + bird2.GetSpecies() + " " + bird2.GetPosition());

        // Prompt user to select a bird (1 or 2), repeat until valid
        int choice = 0;
        boolean validChoice = false;
        while (!validChoice) {
            System.out.println("Select a bird to move (1 or 2):");
            if (input.hasNextInt()) {
                choice = input.nextInt();
                if (choice == 1 || choice == 2) {
                    validChoice = true;
                } else {
                    System.out.println("Invalid choice. Please enter 1 or 2.");
                }
            } else {
                System.out.println("Invalid input. Please enter a number.");
                input.next();
            }
        }

        // Prompt user for direction, repeat until valid
        char direction = ' ';
        boolean validDirection = false;
        while (!validDirection) {
            System.out.println("Enter direction (N, S, E or W):");
            String dirInput = input.next().toUpperCase();
            if (dirInput.length() == 1) {
                direction = dirInput.charAt(0);
                if (direction == 'N' || direction == 'S' || direction == 'E' || direction == 'W') {
                    validDirection = true;
                } else {
                    System.out.println("Invalid direction. Please enter N, S, E or W.");
                }
            } else {
                System.out.println("Invalid direction. Please enter N, S, E or W.");
            }
        }

        // Prompt user for time (0 to 500), repeat until valid
        int minutes = 0;
        boolean validTime = false;
        while (!validTime) {
            System.out.println("Enter time flying in minutes (0 to 500):");
            if (input.hasNextInt()) {
                minutes = input.nextInt();
                if (minutes >= 0 && minutes <= 500) {
                    validTime = true;
                } else {
                    System.out.println("Invalid time. Please enter a value between 0 and 500.");
                }
            } else {
                System.out.println("Invalid input. Please enter a whole number.");
                input.next();
            }
        }

        // Move the chosen bird and output new position
        if (choice == 1) {
            bird1.Move(direction, minutes);
            System.out.println("Bird 1 (" + bird1.GetSpecies() + ") new position: " + bird1.GetPosition());
        } else {
            bird2.Move(direction, minutes);
            System.out.println("Bird 2 (" + bird2.GetSpecies() + ") new position: " + bird2.GetPosition());
        }
    }
}
