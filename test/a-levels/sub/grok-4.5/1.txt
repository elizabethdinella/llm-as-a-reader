```java
public void Move(char direction, int minutes) {
    double distance = (DistancePerHour / 60.0) * minutes;
    if (direction == 'N') {
        YPosition += distance;
    } else if (direction == 'S') {
        YPosition -= distance;
    } else if (direction == 'E') {
        XPosition += distance;
    } else if (direction == 'W') {
        XPosition -= distance;
    }
}
```

```java
import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Bird bird1 = new Bird("Cockatiel", 71.0);
        Bird bird2 = new Bird("Macaw", 56.0);
        Scanner sc = new Scanner(System.in);

        System.out.println(bird1.GetSpecies() + " position: " + bird1.GetPosition());
        System.out.println(bird2.GetSpecies() + " position: " + bird2.GetPosition());

        int choice;
        do {
            System.out.print("Enter 1 to move Cockatiel or 2 to move Macaw: ");
            choice = sc.nextInt();
        } while (choice != 1 && choice != 2);

        char dir;
        do {
            System.out.print("Enter direction (N/S/E/W): ");
            dir = sc.next().toUpperCase().charAt(0);
        } while (dir != 'N' && dir != 'S' && dir != 'E' && dir != 'W');

        int mins;
        do {
            System.out.print("Enter flying time in minutes (0-500): ");
            mins = sc.nextInt();
        } while (mins < 0 || mins > 500);

        if (choice == 1) {
            bird1.Move(dir, mins);
            System.out.println(bird1.GetSpecies() + " new position: " + bird1.GetPosition());
        } else {
            bird2.Move(dir, mins);
            System.out.println(bird2.GetSpecies() + " new position: " + bird2.GetPosition());
        }
        sc.close();
    }
}
```
