import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Bird bird1 = new Bird("Cockatiel", 71.0);
        Bird bird2 = new Bird("Macaw", 56.0);

        Scanner input = new Scanner(System.in);

        // show current status of both birds
        System.out.println("1) " + bird1.getSpecies() + " " + bird1.getPosition());
        System.out.println("2) " + bird2.getSpecies() + " " + bird2.getPosition());

        // choose bird
        int choice;
        do {
            System.out.print("Select the bird to move (1 or 2): ");
            while (!input.hasNextInt()) {          // type validation
                System.out.print("Please enter 1 or 2: ");
                input.next();
            }
            choice = input.nextInt();
        } while (choice != 1 && choice != 2);

        // choose direction
        char direction;
        do {
            System.out.print("Enter direction (N, S, E, W): ");
            String dirStr = input.next().toUpperCase();
            direction = dirStr.length() > 0 ? dirStr.charAt(0) : ' ';
        } while ("NSEW".indexOf(direction) == -1);

        // choose minutes
        int minutes;
        do {
            System.out.print("Enter flying time in minutes (0-500): ");
            while (!input.hasNextInt()) {          // type validation
                System.out.print("Please enter a whole number between 0 and 500: ");
                input.next();
            }
            minutes = input.nextInt();
        } while (minutes < 0 || minutes > 500);

        // move the selected bird
        Bird chosen = (choice == 1) ? bird1 : bird2;
        chosen.move(direction, minutes);

        // output updated position
        System.out.println("Updated position for " + chosen.getSpecies() + ": " + chosen.getPosition());

        input.close();
    }
}
