public void Move(char Direction, int MinutesFlying)
{
    double DistanceTravelled;
    DistanceTravelled = (DistancePerHour / 60) * MinutesFlying;

    switch (Direction)
    {
        case 'N':
            YPosition = YPosition + DistanceTravelled;
            break;
        case 'S':
            YPosition = YPosition - DistanceTravelled;
            break;
        case 'E':
            XPosition = XPosition + DistanceTravelled;
            break;
        case 'W':
            XPosition = XPosition - DistanceTravelled;
            break;
    }
}
