import java.util.Scanner;

public class BirdProgram
{
    public static void main(String[] args)
    {
        Scanner Input = new Scanner(System.in);

        Bird Bird1 = new Bird("Cockatiel", 71.0);
        Bird Bird2 = new Bird("Macaw", 56.0);

        // output current data for both birds
        System.out.println("Bird 1 is a " + Bird1.GetSpecies() + " at position " + Bird1.GetPosition());
        System.out.println("Bird 2 is a " + Bird2.GetSpecies() + " at position " + Bird2.GetPosition());

        // input and validate which bird to move
        int Choice;
        do
        {
            System.out.println("Enter 1 to move the " + Bird1.GetSpecies()
                             + " or 2 to move the " + Bird2.GetSpecies() + ":");
            while (!Input.hasNextInt())
            {
                System.out.println("Invalid input. Please enter 1 or 2:");
                Input.next();
            }
            Choice = Input.nextInt();
            if (Choice != 1 && Choice != 2)
            {
                System.out.println("Invalid choice, you must enter 1 or 2.");
            }
        } while (Choice != 1 && Choice != 2);

        // input and validate the direction
        char Direction;
        do
        {
            System.out.println("Enter the direction of travel (N, S, E or W):");
            Direction = Character.toUpperCase(Input.next().charAt(0));
            if (Direction != 'N' && Direction != 'S' && Direction != 'E' && Direction != 'W')
            {
                System.out.println("Invalid direction, you must enter N, S, E or W.");
            }
        } while (Direction != 'N' && Direction != 'S' && Direction != 'E' && Direction != 'W');

        // input and validate the number of minutes
        int Minutes;
        do
        {
            System.out.println("Enter the number of minutes flying (0 to 500):");
            while (!Input.hasNextInt())
            {
                System.out.println("Invalid input. Please enter a whole number between 0 and 500:");
                Input.next();
            }
            Minutes = Input.nextInt();
            if (Minutes < 0 || Minutes > 500)
            {
                System.out.println("Invalid time, it must be between 0 and 500 minutes.");
            }
        } while (Minutes < 0 || Minutes > 500);

        // move the chosen bird and output its new position
        if (Choice == 1)
        {
            Bird1.Move(Direction, Minutes);
            System.out.println("The " + Bird1.GetSpecies() + " has moved to " + Bird1.GetPosition());
        }
        else
        {
            Bird2.Move(Direction, Minutes);
            System.out.println("The " + Bird2.GetSpecies() + " has moved to " + Bird2.GetPosition());
        }
    }
}
