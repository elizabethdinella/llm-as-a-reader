import java.util.InputMismatchException;
import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Bird cockatiel = new Bird("Cockatiel", 71.0);
        Bird macaw = new Bird("Macaw", 56.0);

        System.out.println("Initial positions:");
        System.out.println(cockatiel.getSpecies() + " - " + cockatiel.getPosition());
        System.out.println(macaw.getSpecies() + " - " + macaw.getPosition());

        Scanner scanner = new Scanner(System.in);

        while (true) {
            try {
                System.out.print("Select a bird to move (1 for Cockatiel, 2 for Macaw): ");
                int choice = scanner.nextInt();
                Bird selectedBird;
                if (choice == 1) {
                    selectedBird = cockatiel;
                } else if (choice == 2) {
                    selectedBird = macaw;
                } else {
                    System.out.println("Invalid choice. Please try again.");
                    continue;
                }

                System.out.print("Enter direction (N, S, E, W): ");
                char direction = scanner.next().charAt(0);

                System.out.print("Enter time flying (minutes): ");
                int minutesFlying = scanner.nextInt();

                selectedBird.move(direction, minutesFlying);

                System.out.println(selectedBird.getSpecies() + " - New position: " + selectedBird.getPosition());
                break;
            } catch (InputMismatchException e) {
                System.out.println("Invalid input. Please try again.");
                scanner.next(); // Clear invalid input
            }
        }

        scanner.close();
    }
}
