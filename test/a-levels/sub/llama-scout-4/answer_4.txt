public void move(char direction, int minutesFlying) {
    double distanceTraveled = (distancePerHour / 60) * minutesFlying;

    switch (direction) {
        case 'N':
            yPosition += distanceTraveled;
            break;
        case 'S':
            yPosition -= distanceTraveled;
            break;
        case 'E':
            xPosition += distanceTraveled;
            break;
        case 'W':
            xPosition -= distanceTraveled;
            break;
    }
}
