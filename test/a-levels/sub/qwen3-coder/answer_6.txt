Scanner scanner = new Scanner(System.in);

System.out.println("Bird 1: " + bird1.GetSpecies() + " - " + bird1.GetPosition());
System.out.println("Bird 2: " + bird2.GetSpecies() + " - " + bird2.GetPosition());

int choice;
do {
    System.out.print("Select bird to move (1 or 2): ");
    choice = scanner.nextInt();
} while (choice != 1 && choice != 2);

char direction;
do {
    System.out.print("Enter direction (N, S, E, W): ");
    direction = scanner.next().charAt(0);
} while (direction != 'N' && direction != 'S' && direction != 'E' && direction != 'W');

int minutes;
do {
    System.out.print("Enter time in minutes (0-500): ");
    minutes = scanner.nextInt();
} while (minutes < 0 || minutes > 500);

if (choice == 1) {
    bird1.Move(direction, minutes);
    System.out.println("Bird 1 new position: " + bird1.GetPosition());
} else {
    bird2.Move(direction, minutes);
    System.out.println("Bird 2 new position: " + bird2.GetPosition());
}
