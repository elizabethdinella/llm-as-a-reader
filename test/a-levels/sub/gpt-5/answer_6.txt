import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);

        Bird bird1 = new Bird("Cockatiel", 71.0);
        Bird bird2 = new Bird("Macaw", 56.0);

        System.out.println("Current birds and positions:");
        System.out.println("1) " + bird1.GetSpecies() + " - " + bird1.GetPosition());
        System.out.println("2) " + bird2.GetSpecies() + " - " + bird2.GetPosition());

        int selection;
        while (true) {
            System.out.print("Select a bird to move (enter 1 for " + bird1.GetSpecies() + " or 2 for " + bird2.GetSpecies() + "): ");
            String line = sc.nextLine().trim();
            try {
                selection = Integer.parseInt(line);
                if (selection == 1 || selection == 2) {
                    break;
                }
            } catch (NumberFormatException e) {
                // fall through to invalid message
            }
            System.out.println("Invalid selection. Please enter 1 or 2.");
        }

        Bird chosen = (selection == 1) ? bird1 : bird2;

        char dir;
        while (true) {
            System.out.print("Enter direction (N for north, S for south, E for east, W for west): ");
            String s = sc.nextLine().trim().toUpperCase();
            if (s.length() == 1 && "NSEW".indexOf(s.charAt(0)) >= 0) {
                dir = s.charAt(0);
                break;
            }
            System.out.println("Invalid direction. Please enter N, S, E, or W.");
        }

        int minutes;
        while (true) {
            System.out.print("Enter minutes of flight (0 to 500 inclusive): ");
            String line = sc.nextLine().trim();
            try {
                minutes = Integer.parseInt(line);
                if (minutes >= 0 && minutes <= 500) {
                    break;
                }
            } catch (NumberFormatException e) {
                // fall through to invalid message
            }
            System.out.println("Invalid input. Please enter an integer from 0 to 500.");
        }

        chosen.Move(dir, minutes);

        System.out.println("Updated position for " + chosen.GetSpecies() + ": " + chosen.GetPosition());

        sc.close();
    }
}
