public void move(char direction, int minutes) {
    double distance = (distancePerHour / 60.0) * minutes;
    switch (direction) {
        case 'N':
            yPosition += distance;
            break;
        case 'S':
            yPosition -= distance;
            break;
        case 'E':
            xPosition += distance;
            break;
        case 'W':
            xPosition -= distance;
            break;
    }
}
