public int moreHistoryThanMathAbsences() {
    HashMap<String, CourseRecord> historyMap = new HashMap<>();
    for (CourseRecord record : historyList) {
        historyMap.put(record.getStudentID(), record);
    }
    int count = 0;
    for (CourseRecord mathRecord : mathList) {
        String studentId = mathRecord.getStudentID();
        if (historyMap.containsKey(studentId)) {
            CourseRecord historyRecord = historyMap.get(studentId);
            int historyAbsences = historyRecord.getAbsences();
            int mathAbsences = mathRecord.getAbsences();
            if (historyAbsences < mathAbsences) {
                count++;
            }
        }
    }
    return count;
}
