public int getPointsForRow(int targetRow)
{
    int sumPoints = 0;
    boolean allSameColor = true;

    String firstColor = board[0].getColor();

    for (int col = 0; col < board[targetRow].length; col++)
    {
        sumPoints += board[col].getPoints();
        if (!board[col].getColor().equals(firstColor))
        {
            allSameColor = false;
        }
    }

    if (allSameColor)
    {
        return 2 * sumPoints;
    }
    else
    {
        return sumPoints;
    }
}
