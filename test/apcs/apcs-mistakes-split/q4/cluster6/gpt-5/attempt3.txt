public int getPointsForRow(int targetRow)
{
    Space[] row = board[targetRow];
    int sum = 0;
    String firstColor = row[0].getColor();
    boolean allSameColor = true;

    for (int c = 1; c < row.length; c++)
    {
        sum += row[c].getPoints();
        if (!row[c].getColor().equals(firstColor))
        {
            allSameColor = false;
        }
    }

    return allSameColor ? 2 * sum : sum;
}
