public class Bottle {
    private double capacity;
    private double amount;

    public Bottle(double capacity) {
        this.capacity = capacity;
        this.amount = capacity;
    }

    public double updateAmount(double remove) {
        amount -= remove;
        if (amount < 0.25 * capacity) {
            amount = capacity;
        }
        return amount;
    }
}
