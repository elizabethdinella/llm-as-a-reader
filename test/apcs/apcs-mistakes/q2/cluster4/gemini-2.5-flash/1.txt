public class Bottle {
    private double capacity;
    private double currentAmount;

    public Bottle(double capacity) {
        this.capacity = capacity;
        this.currentAmount = capacity; // Filled to capacity upon construction
    }

    public double updateAmount(double amountRemoved) {

        // Check if the remaining amount is less than 25% of the capacity
        if (this.currentAmount < (0.25 * this.capacity)) {

    }
