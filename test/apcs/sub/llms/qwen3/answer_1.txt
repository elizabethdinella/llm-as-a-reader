public Account(String requestedName) {
    String current = requestedName;
    if (isAvailable(current)) {
        this.username = current;
    } else {
        int suffix = 1;
        while (true) {
            current = requestedName + suffix;
            if (isAvailable(current)) {
                break;
            }
            suffix++;
        }
        this.username = current;
    }
}
