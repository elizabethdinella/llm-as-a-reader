public int moreHistoryThanMathAbsences() {
    int count = 0;
    HashMap<String, Integer> mathAbsencesMap = new HashMap<>();
    for (CourseRecord cr : mathList) {
        mathAbsencesMap.put(cr.getStudentID(), cr.getAbsences());
    }
    for (CourseRecord cr : historyList) {
        String id = cr.getStudentID();
        if (mathAbsencesMap.containsKey(id)) {
            int historyAbs = cr.getAbsences();
            int mathAbs = mathAbsencesMap.get(id);
            if (historyAbs >= mathAbs) {
                count++;
            }
        }
    }
    return count;
}
