public class Bottle
{
    private double capacity;
    private double amount;

    public Bottle(double cap)
    {
        capacity = cap;
        amount = 0;
    }

    public double updateAmount(double used)
    {
        amount -= used;
        if (amount < 0.25 * capacity)
        {
            amount = capacity;
        }
        return amount;
    }
}
